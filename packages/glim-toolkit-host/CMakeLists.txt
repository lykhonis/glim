if(APPLE)
  set(_glim_host_seat src/SeatViewApple.mm)
else()
  set(_glim_host_seat src/SeatViewStub.cpp)
endif()

add_library(glim-toolkit-host STATIC src/Host.cpp src/Drive.cpp src/Probe.cpp src/TestApp.cpp ${_glim_host_seat})
add_library(glim::toolkit-host ALIAS glim-toolkit-host)
target_include_directories(glim-toolkit-host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(glim-toolkit-host PUBLIC glim::toolkit glim::paint-gpu glim::image)
if(TARGET glim::gpu-metal)
  target_link_libraries(glim-toolkit-host PUBLIC glim::gpu-metal)
endif()
if(TARGET glim::gpu-vulkan)
  target_link_libraries(glim-toolkit-host PUBLIC glim::gpu-vulkan)
endif()
if(TARGET glim::gpu-webgpu)
  target_link_libraries(glim-toolkit-host PUBLIC glim::gpu-webgpu)
endif()
if(TARGET glim::shell)
  target_link_libraries(glim-toolkit-host PUBLIC glim::shell)
endif()
glim_target_defaults(glim-toolkit-host)
if(APPLE)
  set_source_files_properties(src/SeatViewApple.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
endif()

add_library(glim-toolkit-host-cpu STATIC src/Host.cpp src/Drive.cpp src/Probe.cpp src/TestApp.cpp ${_glim_host_seat})
add_library(glim::toolkit-host-cpu ALIAS glim-toolkit-host-cpu)
target_include_directories(glim-toolkit-host-cpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(glim-toolkit-host-cpu PUBLIC glim::toolkit glim::paint-cpu glim::image)
if(TARGET glim::shell)
  target_link_libraries(glim-toolkit-host-cpu PUBLIC glim::shell)
endif()
target_compile_definitions(glim-toolkit-host-cpu PUBLIC GLIM_SOFTWARE=1 GLIM_EMBED=0)
glim_target_defaults(glim-toolkit-host-cpu)
if(APPLE)
  set_source_files_properties(src/SeatViewApple.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
endif()
