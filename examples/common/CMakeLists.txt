# Headless GPU-vs-CPU golden harness for the build's GPU backend (Vulkan or Metal).
if(GLIM_HOST_TESTS AND GLIM_GPU MATCHES "^(vulkan|metal)$")
    add_library(glim-gpu-golden STATIC GpuGolden.cpp)
    target_include_directories(glim-gpu-golden PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(glim-gpu-golden PUBLIC glim::glim glim::image)
    glim_test_defaults(glim-gpu-golden)
    if(GLIM_GPU STREQUAL "metal")
        target_compile_definitions(glim-gpu-golden PRIVATE GLIM_GOLDEN_METAL=1)
    endif()
endif()

# glim_add_gpu_golden(<example> GOLDEN <png> SOURCES <files...> [LIBS <libs...>])
# Adds test glim-<example>-<gpu>-golden; no-op without a headless-capable GPU.
function(glim_add_gpu_golden example)
    cmake_parse_arguments(ARG "" "GOLDEN" "SOURCES;LIBS" ${ARGN})
    if(NOT TARGET glim-gpu-golden)
        return()
    endif()
    set(_t glim-${example}-${GLIM_GPU}-golden)
    add_executable(${_t} ${ARG_SOURCES})
    target_include_directories(${_t} PRIVATE src)
    target_link_libraries(${_t} PRIVATE glim-gpu-golden ${ARG_LIBS})
    glim_test_defaults(${_t})
    target_compile_definitions(${_t} PRIVATE GLIM_GOLDEN=1)
    add_test(NAME ${_t} COMMAND ${_t} "${ARG_GOLDEN}")
endfunction()
