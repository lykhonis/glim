cmake_minimum_required(VERSION 3.22)
project(glim VERSION 0.1.0 LANGUAGES C CXX)
if(APPLE)
    enable_language(OBJC)
    enable_language(OBJCXX)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# Help editors that look at the source root (gitignored).
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
else()
    file(CREATE_LINK
        "${CMAKE_BINARY_DIR}/compile_commands.json"
        "${CMAKE_SOURCE_DIR}/compile_commands.json"
        SYMBOLIC COPY_ON_ERROR)
endif()

enable_testing()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(GlimCommon)

add_subdirectory(packages/glim-core)
add_subdirectory(packages/glim-image)
if(GLIM_GUEST)
    add_subdirectory(packages/glim-wire)
    add_subdirectory(packages/glim-transport)
endif()
add_subdirectory(packages/glim-gpu)
add_subdirectory(packages/glim-shell)

if(APPLE AND GLIM_GPU STREQUAL "metal")
    add_subdirectory(packages/glim-gpu-metal)
elseif(GLIM_GPU STREQUAL "vulkan")
    add_subdirectory(packages/glim-gpu-vulkan)
elseif(GLIM_GPU STREQUAL "webgpu")
    add_subdirectory(packages/glim-gpu-webgpu)
endif()
if(NOT TARGET glim-vulkan-shaders)
    add_custom_target(glim-vulkan-shaders)
endif()

add_subdirectory(packages/glim-paint)
if(GLIM_GUEST)
    add_subdirectory(packages/glim-guest)
    add_subdirectory(packages/glim-guest-host)
endif()
add_subdirectory(packages/glim-embed)
add_subdirectory(packages/glim-toolkit)

if(APPLE AND GLIM_SHELL STREQUAL "macos")
    enable_language(OBJCXX)
    add_subdirectory(packages/glim-shell-macos)
elseif(APPLE AND (GLIM_SHELL STREQUAL "ios" OR GLIM_SHELL STREQUAL "tvos"))
    enable_language(OBJCXX)
    add_subdirectory(packages/glim-shell-ios)
elseif(GLIM_SHELL STREQUAL "wayland")
    add_subdirectory(packages/glim-shell-wayland)
elseif(GLIM_SHELL STREQUAL "web")
    add_subdirectory(packages/glim-shell-web)
elseif(GLIM_SHELL STREQUAL "custom")
    add_subdirectory(packages/glim-shell-custom)
elseif(ANDROID AND GLIM_SHELL STREQUAL "android")
    add_subdirectory(packages/glim-shell-android)
endif()

if(NOT GLIM_SHELL STREQUAL "none")
    add_subdirectory(examples)
endif()

add_library(glim INTERFACE)
add_library(glim::glim ALIAS glim)
if(GLIM_SOFTWARE)
    if(GLIM_EMBED)
        target_link_libraries(glim INTERFACE glim::core glim::paint glim::paint-embed-cpu glim::embed)
    else()
        target_link_libraries(glim INTERFACE glim::core glim::paint glim::paint-cpu glim::embed)
    endif()
elseif(GLIM_EMBED)
    target_link_libraries(glim INTERFACE glim::core glim::gpu glim::paint glim::paint-embed-gpu glim::embed)
else()
    target_link_libraries(glim INTERFACE glim::core glim::gpu glim::paint glim::paint-gpu glim::embed)
endif()
if(TARGET glim::gpu-metal)
    target_link_libraries(glim INTERFACE glim::gpu-metal)
endif()
if(TARGET glim::gpu-vulkan)
    target_link_libraries(glim INTERFACE glim::gpu-vulkan)
endif()
if(TARGET glim::gpu-webgpu)
    target_link_libraries(glim INTERFACE glim::gpu-webgpu)
endif()
if(TARGET glim::shell)
    target_link_libraries(glim INTERFACE glim::shell)
endif()

if(NOT GLIM_SHELL STREQUAL "none")
    add_subdirectory(packages/glim-toolkit-host)
endif()

option(GLIM_INSTALL "Install the glim CMake package" ${PROJECT_IS_TOP_LEVEL})
if(GLIM_INSTALL)
    include(GlimInstall)
endif()
